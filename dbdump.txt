CREATE OR REPLACE FUNCTION "public"."can_manage_medical_record"("p_medical_record_id" "uuid") RETURNS boolean
    LANGUAGE "plpgsql" SECURITY DEFINER
    AS $$
DECLARE
  is_allowed boolean;
BEGIN
  IF auth.role() != 'authenticated' THEN
    RETURN FALSE;
  END IF;
  
  -- For INSERT checks where the record doesn't exist yet, we can't check its properties.
  -- In this case, we rely on the broader 'is_content_creator' role check in the policy itself.
  -- For UPDATE/DELETE, we can check the existing record.
  IF TG_OP = 'INSERT' THEN
     RETURN public.is_content_creator();
  END IF;

  SELECT (
    public.is_admin() OR
    (p.examining_doctor_id = auth.uid()) -- The examining doctor
  ) INTO is_allowed
  FROM public.medical_records p
  WHERE p.id = p_medical_record_id;

  RETURN COALESCE(is_allowed, FALSE);
EXCEPTION
  WHEN OTHERS THEN
    RETURN FALSE;
END;
$$;


ALTER FUNCTION "public"."can_manage_medical_record"("p_medical_record_id" "uuid") OWNER TO "postgres";


CREATE OR REPLACE FUNCTION "public"."can_view_medical_record"("record_id" "uuid") RETURNS boolean
    LANGUAGE "plpgsql" SECURITY DEFINER
    AS $$
DECLARE
  is_allowed boolean;
BEGIN
  IF auth.role() != 'authenticated' THEN
    RETURN FALSE;
  END IF;

  SELECT (
    public.is_admin() OR
    (p.patient_id = auth.uid()) OR -- The patient themselves
    (p.examining_doctor_id = auth.uid()) -- The examining doctor
  ) INTO is_allowed
  FROM public.medical_records p
  WHERE p.id = record_id;

  RETURN COALESCE(is_allowed, FALSE);
EXCEPTION
  WHEN OTHERS THEN
    RETURN FALSE;
END;
$$;


ALTER FUNCTION "public"."can_view_medical_record"("record_id" "uuid") OWNER TO "postgres";


CREATE OR REPLACE FUNCTION "public"."demote_role_on_doctor_delete"() RETURNS "trigger"
    LANGUAGE "plpgsql" SECURITY DEFINER
    AS $$
BEGIN
  UPDATE public.patients
  SET role = 'customer'
  WHERE id = OLD.id;
  RETURN OLD;
END;
$$;


ALTER FUNCTION "public"."demote_role_on_doctor_delete"() OWNER TO "postgres";


CREATE OR REPLACE FUNCTION "public"."demote_user_on_doctor_delete"() RETURNS "trigger"
    LANGUAGE "plpgsql" SECURITY DEFINER
    AS $$
BEGIN
  UPDATE public.patients
  SET role = 'customer'
  WHERE id = OLD.id AND role = 'doctor'; -- Only demote if their current role is 'doctor'
  RETURN OLD;
END;
$$;

CREATE OR REPLACE FUNCTION "public"."handle_new_user"() RETURNS "trigger"
    LANGUAGE "plpgsql" SECURITY DEFINER
    AS $$
BEGIN
  INSERT INTO public.patients (
    id,
    name,
    dob,
    phone,
    role,
    avatar_path
  )
  VALUES (
    NEW.id,
    COALESCE(NEW.raw_user_meta_data->>'full_name', NEW.raw_user_meta_data->>'name', 'Người dùng mới'),
    COALESCE(NEW.raw_user_meta_data->>'dob', '1990-01-01'),
    COALESCE(NEW.raw_user_meta_data->>'phone', '0000000000'),
    'customer',
    NEW.raw_user_meta_data->>'avatar_url'
  );
  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION "public"."is_admin"() RETURNS boolean
    LANGUAGE "plpgsql" SECURITY DEFINER
    SET "search_path" TO 'public'
    AS $$
BEGIN
    IF auth.role() = 'authenticated' THEN
        RETURN EXISTS (SELECT 1 FROM public.patients WHERE id = auth.uid() AND role IN ('admin', 'master_admin'));
    END IF;
    RETURN FALSE;
EXCEPTION
    WHEN OTHERS THEN
        RETURN FALSE;
END;
$$;


ALTER FUNCTION "public"."is_admin"() OWNER TO "postgres";
